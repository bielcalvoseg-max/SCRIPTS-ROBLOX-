local function Notify(title, content, duration)
    Rayfield:Notify({

local function GetPlayer(str)
    if not str or str == "" then return nil end
    str = str:lower()
    if str == "me" then return LocalPlayer end
    if str == "random" then
        local plrs = Players:GetPlayers()
        return plrs[math.random(1, #plrs)]
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():sub(1, #str) == str or p.DisplayName:lower():sub(1, #str) == str then
            return p
        end
    end
    return nil
end

local function GetCharacter(plr)
    plr = plr or LocalPlayer
    return plr.Character or plr.CharacterAdded:Wait()
end

local function GetRoot(plr)
    local char = GetCharacter(plr)
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
end

local function GetHumanoid(plr)
    local char = GetCharacter(plr)
    return char and char:FindFirstChildOfClass("Humanoid")
end

-- 4. Sistema de Fly
local flyBodyVel, flyBodyGyro
local function ToggleFly(enabled)
    State.Fly = enabled
    local root = GetRoot()
    if not root then return end

    if enabled then
        flyBodyVel = Instance.new("BodyVelocity")
        flyBodyVel.Velocity = Vector3.zero
        flyBodyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        flyBodyVel.Name = "Admin_FlyVel"
        flyBodyVel.Parent = root

        flyBodyGyro = Instance.new("BodyGyro")
        flyBodyGyro.P = 9e4
        flyBodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        flyBodyGyro.CFrame = root.CFrame
        flyBodyGyro.Name = "Admin_FlyGyro"
        flyBodyGyro.Parent = root

        task.spawn(function()
            while State.Fly and root and root.Parent do
                local camCF = Camera.CFrame
                local moveDir = Vector3.zero

                if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

                if moveDir.Magnitude > 0 then
                    flyBodyVel.Velocity = moveDir.Unit * State.FlySpeed
                else
                    flyBodyVel.Velocity = Vector3.zero
                end
                flyBodyGyro.CFrame = camCF
                RunService.RenderStepped:Wait()
            end
            if flyBodyVel then flyBodyVel:Destroy() end
            if flyBodyGyro then flyBodyGyro:Destroy() end
        end)
    else
        if flyBodyVel then flyBodyVel:Destroy() end
        if flyBodyGyro then flyBodyGyro:Destroy() end
    end
end

-- 6. Orbit & Follow
local orbitAngle = 0
RunService.RenderStepped:Connect(function()
    local root = GetRoot()
    if not root then return end

    if State.OrbitTarget and State.OrbitTarget.Character then
        local tRoot = GetRoot(State.OrbitTarget)
        if tRoot then
            orbitAngle = orbitAngle + (State.OrbitSpeed * 0.05)
            local offset = Vector3.new(math.cos(orbitAngle) * State.OrbitDist, 0, math.sin(orbitAngle) * State.OrbitDist)
            root.CFrame = CFrame.new(tRoot.Position + offset, tRoot.Position)
        end
    end

    if State.FollowTarget and State.FollowTarget.Character then
        local tRoot = GetRoot(State.FollowTarget)
        if tRoot then
            local hum = GetHumanoid()
            if hum then hum:MoveTo(tRoot.Position) end
        end
    end

    if State.Spin and root then
        root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(State.SpinSpeed), 0)
    end
end)

local origAmbient = Lighting.Ambient
local origOutdoor = Lighting.OutdoorAmbient
local origBrightness = Lighting.Brightness
local origClock = Lighting.ClockTime
local origShadows = Lighting.GlobalShadows

local function SetFullbright(val)
    State.Fullbright = val
    if val then
        Lighting.Ambient = Color3.fromRGB(255, 255, 255)
        Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
        Lighting.Brightness = 2
        Lighting.GlobalShadows = false
    else
        Lighting.Ambient = origAmbient
        Lighting.OutdoorAmbient = origOutdoor
        Lighting.Brightness = origBrightness
        Lighting.GlobalShadows = origShadows
    end
end

-- 8. ESP Universal
local espFolder = Instance.new("Folder")
espFolder.Name = "Admin_ESP_Holder"
espFolder.Parent = Camera

local function UpdateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local char = p.Character
            local hl = espFolder:FindFirstChild("HL_" .. p.Name)
            if State.ESP or State.Chams then
                if not hl then
                    hl = Instance.new("Highlight")
                    hl.Name = "HL_" .. p.Name
                    hl.Adornee = char
                    hl.FillColor = p.TeamColor and p.TeamColor.Color or Color3.fromRGB(255, 50, 50)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = State.Chams and 0.3 or 0.8
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    hl.Parent = espFolder
                else
                    hl.Adornee = char
                    hl.Enabled = true
                end
            else
                if hl then hl.Enabled = false end
            end
        end
    end
end

task.spawn(function()
    while true do
        if State.ESP or State.Chams then UpdateESP() end
        task.wait(1)
    end
end)

    local c = GetCharacter()
    if c then
        for _, obj in ipairs(c:GetChildren()) do
            if obj:IsA("CharacterMesh") then obj:Destroy() end
        end
    end
end)

-- Função de Execução de Texto
local function ExecuteAdminCommand(text)
    if not text or text == "" then return end
    if text:sub(1, 1) == ";" or text:sub(1, 1) == "/" or text:sub(1, 1) == "!" then
        text = text:sub(2)
    end
    local parts = text:split(" ")
    local cmdName = parts[1]:lower()
    table.remove(parts, 1)

    if Commands[cmdName] then
        local success, err = pcall(function()
            Commands[cmdName].Func(parts)
        end)
        if not success then
            Notify("Erro de Comando", tostring(err), 3)
        end
    else
        Notify("Comando Desconhecido", "O comando '" .. cmdName .. "' não existe.", 2)
    end
end
